# Wait for our database to become available.
#[root@calycopis-dev]

    postgreswait()
        {
        for ((i = 1; i <= 4; i++))
        do
            if pg_isready \
                --host "${POSTGRES_HOSTNAME:?}" \
                --port "${POSTGRES_PORT:?}" \
                --dbname "${POSTGRES_DBNAME:?}"
            then
                echo "[$(date)] database is ready"
                return 0
            fi
            echo "[$(date)] waiting for database to start (${i}/10)."
            sleep 10
        done
        echo "[$(date)] database is NOT ready"
            return 1
        }

    postgreswait
