# Change the Podman subnet ranges to avoid those used by our LAN.
# https://chatgpt.com/share/6aa8d9b0-5ddc-83eb-944a-c1edb42fbad5
#[user@desktop]

    cat > "${HOME}/.config/containers/containers.conf" << EOF
#
# Subnet configuration to avoid 10/8 which is used by our LAN.
# https://chatgpt.com/share/6aa8d9b0-5ddc-83eb-944a-c1edb42fbad5

[network]

default_subnet = "172.29.0.0/16"

default_subnet_pools = [
    {"base" = "172.30.0.0/16", "size" = 24},
    ]
EOF

    systemctl --user daemon-reload

    podman network inspect podman | jq '.[0].subnets.[0]'

    >   {
    >     "subnet": "172.29.0.0/16",
    >     "gateway": "172.29.0.1"
    >   }


    podman network create test-network

    podman network inspect test-network | jq '.[0].subnets.[0]'

    >   {
    >     "subnet": "172.30.0.0/24",
    >     "gateway": "172.30.0.1"
    >   }

    podman network rm test-network

# -----------------------------------------------------
# Delete everything from Podman.
#[user@desktop]

    source "${HOME}/calycopis.env"
    source "${CALYCOPIS_CODE}/calycopis.vars"

    podman pod stop   "${CALYCOPIS_POD_NAME:?}"
    podman pod rm     "${CALYCOPIS_POD_NAME:?}"
    podman network rm "${CALYCOPIS_NET_NAME:?}"

    podman rm $(
        podman ps -aq
        )

    podman volume rm $(
        podman volume ls -q
        )

# -----------------------------------------------------
# Create our Podman network and Pod.
#[user@desktop]

    source "${HOME}/calycopis.env"
    source "${CALYCOPIS_CODE}/calycopis.vars"

    podman network create \
        "${CALYCOPIS_NET_NAME:?}"

    podman pod create \
        --name "${CALYCOPIS_POD_NAME:?}" \
        --network "${CALYCOPIS_NET_NAME:?}" \
        --publish 8082:8082 \
        --publish 3081:3081


# -----------------------------------------------------
# Configure the broker-config volume.
#[user@desktop]

    source "${HOME}/calycopis.env"
    source "${CALYCOPIS_CODE}/calycopis.vars"

    podman run \
        --rm \
        --pod "${CALYCOPIS_POD_NAME:?}" \
        --volume "${CALYCOPIS_BROKER_CONFIG_VOLUME:?}:${CALYCOPIS_BROKER_CONFIG_PATH}" \
        --volume "${CALYCOPIS_CODE}/bin:${CALYCOPIS_BROKER_OPT:?}/bin" \
        --env-file "${CALYCOPIS_CODE}/calycopis.vars" \
        localhost/calycopis/developer-tools:2026.09.14 \
            "${CALYCOPIS_BROKER_OPT:?}/bin/config-broker.sh"

# Start our PostgreSQL database.
#[user@desktop]

    source "${HOME}/calycopis.env"
    source "${CALYCOPIS_CODE}/calycopis.vars"

    podman run \
        --rm \
        --detach \
        --pod "${CALYCOPIS_POD_NAME:?}" \
        --name "${CALYCOPIS_DATABASE_HOSTNAME}" \
        --expose "${CALYCOPIS_DATABASE_PORT:?}" \
        --env "POSTGRES_DB=${CALYCOPIS_DATABASE_NAME:?}" \
        --env "POSTGRES_USER_FILE=${CALYCOPIS_DATABASE_CONFIG_PATH:?}/pgusername" \
        --env "POSTGRES_PASSWORD_FILE=${CALYCOPIS_DATABASE_CONFIG_PATH:?}/pgpassword" \
        --volume "${CALYCOPIS_DATABASE_CONFIG_VOLUME:?}:${CALYCOPIS_DATABASE_CONFIG_PATH}" \
        "docker.io/library/postgres:latest"

# Build and run our broker service.
#[user@desktop]

        --rm \
        --tty \
        --interactive \
        --pod "${CALYCOPIS_POD_NAME:?}" \
        --name "${CALYCOPIS_BROKER_HOSTNAME:?}" \
        --volume "${CALYCOPIS_BROKER_LOGS}" \
        --volume "${CALYCOPIS_BROKER_CONFIG_VOLUME:?}:${CALYCOPIS_BROKER_CONFIG_PATH}" \
        --volume "${CALYCOPIS_ROOT:?}:/Calycopis:rw,z" \
        --volume "${CALYCOPIS_CODE:?}/calycopis.env:/root/calycopis.env:rw,z" \
        --volume "${HOME:?}/.m2:/root/.m2:rw,z" \
        --env "CONTAINER_HOST=unix:///run/podman/podman.sock" \
        --volume "${XDG_RUNTIME_DIR:?}/podman/podman.sock:/run/podman/podman.sock:rw,z" \
        --env-file "${CALYCOPIS_CODE}/calycopis.vars" \
        localhost/calycopis/developer-tools:2026.09.14 \
