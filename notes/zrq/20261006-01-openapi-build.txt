# Create our dev Pod.
#[user@desktop]

    podman pod create \
        --replace \
        --name calycopis-dev-pod


# -----------------------------------------------------
# Run an instance of our development environment.
#[user@desktop]

    # Using CALYCOPIS_CODE and TREBULA_CODE environment variables to locate the code.
    # Hopefully this will make things more portable.

    source "${HOME:?}/calycopis.env"
    source "${CALYCOPIS_CODE}/bin/container-host.sh"

    podman run \
        --rm \
        --tty \
        --interactive \
        --pod  calycopis-dev-pod \
        --name calycopis-dev-container \
        --volume "${HOME:?}/.m2:/root/.m2:rw,Z" \
        --env    "CALYCOPIS_CODE=/Calycopis/Calycopis-broker" \
        --volume "${CALYCOPIS_CODE:?}:/Calycopis/Calycopis-broker:rw,Z" \
        --env    "TREBULA_CODE=/Calycopis/Calycopis-openapi" \
        --volume "${TREBULA_CODE:?}:/Calycopis/Calycopis-openapi:rw,Z" \
        ghcr.io/ivoa/calycopis/developer-tools:2026.09.17 \
