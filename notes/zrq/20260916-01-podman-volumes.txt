# Delete everything from Podman.
#[user@desktop]

    source "${HOME}/calycopis.env"
    source "${CALYCOPIS_CODE}/calycopis.vars"

    podman pod stop   "${CALYCOPIS_POD_NAME:?}"
    podman pod rm     "${CALYCOPIS_POD_NAME:?}"
    podman network rm "${CALYCOPIS_NET_NAME:?}"

    podman rm $(
        podman ps -aq
        )

    podman volume rm $(
        podman volume ls -q
        )


# -----------------------------------------------------
# Build our broker service.
#[user@desktop]

    source "${HOME}/calycopis.env"
    source "${CALYCOPIS_CODE}/calycopis.vars"
    source "${CALYCOPIS_CODE}/bin/container-host.sh"

    export CONTAINER_PATH=${HOST_CONTAINER_PATH}
    export CONTAINER_HOST=${HOST_CONTAINER_HOST}

    pushd "${CALYCOPIS_CODE}"
        source bin/versions.sh config.yaml
        pushd  java
            ./mvnw clean spring-boot:build-image
